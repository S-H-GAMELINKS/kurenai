# CRuby as a static library for the kernel.
#
# Configured by the superbuild as a child project, so that it can use the
# bare-metal toolchain file and still write ordinary CMake. See the NOTE in the
# top-level CMakeLists.txt about why ExternalProject outputs cannot be consumed
# in the same project that builds them.
#
# The source list is written out rather than globbed. CRuby gains and loses
# files between releases, and a glob would quietly pull in whatever appeared;
# an explicit list turns that into a build error that names the file.

cmake_minimum_required(VERSION 3.25)
project(cruby_cross LANGUAGES C ASM)

foreach(v CRUBY_SRC CRUBY_HOST_BUILD)
  if(NOT ${v})
    message(FATAL_ERROR "cruby_build: -D${v}=... is required")
  endif()
endforeach()

# Extensions linked in statically. Four of them are for irb's sake: Reline
# drives io/console for raw-mode input and io/wait to poll it, builds its output
# through StringIO, and IRB::RubyLex uses Ripper::Lexer to decide whether an
# input line is complete. ripper's sources are generated by the host stage; the
# object list matches $objs in its extconf.rb.
set(EXT_SOURCES
  ${CRUBY_SRC}/ext/io/console/console.c
  ${CRUBY_SRC}/ext/io/wait/wait.c
  ${CRUBY_SRC}/ext/stringio/stringio.c
  ${CRUBY_SRC}/ext/ripper/ripper.c
  ${CRUBY_SRC}/ext/ripper/ripper_init.c
  ${CRUBY_SRC}/ext/ripper/eventids1.c
  ${CRUBY_SRC}/ext/ripper/eventids2.c)

# ext/socket, for the HTTP server. All fifteen objects, matching $objs in its
# extconf.rb, and not because all fifteen are wanted: rsock_init_socket_init
# calls rsock_init_unixsocket, rsock_init_sockssocket, rsock_init_ancdata and
# rsock_init_sockifaddr unconditionally, so leaving an object out is a link
# error rather than a smaller build. What is actually left out is decided in
# cruby_build/socket_extconf.h, which empties those files from the inside and
# leaves their init functions as stubs.
#
# getaddrinfo.c and getnameinfo.c are absent: those are ext/socket's own
# implementations, compiled only when the platform has none. musl has both.
#
# constdefs.c is generated by the host stage, like ripper's sources.
set(SOCKET_SOURCES
  ${CRUBY_SRC}/ext/socket/init.c
  ${CRUBY_SRC}/ext/socket/constants.c
  ${CRUBY_SRC}/ext/socket/basicsocket.c
  ${CRUBY_SRC}/ext/socket/socket.c
  ${CRUBY_SRC}/ext/socket/ipsocket.c
  ${CRUBY_SRC}/ext/socket/tcpsocket.c
  ${CRUBY_SRC}/ext/socket/tcpserver.c
  ${CRUBY_SRC}/ext/socket/sockssocket.c
  ${CRUBY_SRC}/ext/socket/udpsocket.c

  ${CRUBY_SRC}/ext/socket/unixserver.c
  ${CRUBY_SRC}/ext/socket/option.c
  ${CRUBY_SRC}/ext/socket/ancdata.c
  ${CRUBY_SRC}/ext/socket/raddrinfo.c
  ${CRUBY_SRC}/ext/socket/ifaddr.c)

set_source_files_properties(${CRUBY_SRC}/ext/stringio/stringio.c PROPERTIES

  COMPILE_OPTIONS "-include;${CMAKE_CURRENT_SOURCE_DIR}/io_nonblock_extconf.h")

  ${CRUBY_SRC}/ext/date/date_core.c
  ${CRUBY_SRC}/ext/date/date_parse.c
  ${CRUBY_SRC}/ext/date/date_strftime.c
  ${CRUBY_SRC}/ext/date/date_strptime.c

# json writes no extconf.h; its extconf.rb puts what have_func found straight
# into $defs, so these are -D flags rather than an -include. Each function is
# present in CRuby 4.0. JSON_ENABLE_SIMD is left undefined, which is the
# portable path: simd/conf.rb defines it only after compiling a test program
# against the host's x86intrin.h, and there is no such test to run here.
set_source_files_properties(${JSON_SOURCES} PROPERTIES
  COMPILE_DEFINITIONS "HAVE_RB_ENC_INTERNED_STR=1;HAVE_RB_STR_TO_INTERNED_STR=1;HAVE_RB_HASH_NEW_CAPA=1;HAVE_RB_HASH_BULK_INSERT=1")
set_property(SOURCE ${CRUBY_SRC}/ext/json/generator/generator.c APPEND PROPERTY
  COMPILE_DEFINITIONS JSON_GENERATOR)

# NOTE:
# ext/socket cannot be given its extconf.h with -include, the way the three
# above are. rubysocket.h opens with an unconditional
#
#     #include RUBY_EXTCONF_H
#
# so the macro has to exist and has to name a header the compiler can find;
# -include would satisfy the definitions but not that line.
#
# The two IMPL macros are stated rather than inferred. Leaving out
# HAVE_PTHREAD_CREATE already selects FAST_FALLBACK_INIT_INETSOCK_IMPL 0, but
# it selects GETADDRINFO_IMPL *1*, which resolves names with the GVL released.
# There is one thread and nothing to release it to, so 0 -- call getaddrinfo
# directly -- is both simpler and the only variant that cannot want a thread.
# Saying both out loud also means a stray define cannot quietly resurrect
# raddrinfo.c's pthread_create or ipsocket.c's Happy Eyeballs path, which
# builds a thread, a pipe and a select loop for every connect.
set_source_files_properties(${SOCKET_SOURCES} PROPERTIES
  COMPILE_DEFINITIONS "RUBY_EXTCONF_H=\"socket_extconf.h\";GETADDRINFO_IMPL=0;FAST_FALLBACK_INIT_INETSOCK_IMPL=0"
  INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR};${CRUBY_SRC}/ext/socket")

set_target_properties(ruby-static PROPERTIES
  OUTPUT_NAME ruby-static
  PREFIX "lib")

target_include_directories(ruby-static PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}                  # verconf.h
  ${CMAKE_CURRENT_SOURCE_DIR}/include          # the hand-written ruby/config.h
  ${CRUBY_HOST_BUILD}
  ${CRUBY_HOST_BUILD}/.ext/include
  ${CRUBY_SRC}/include
  ${CRUBY_SRC}
  ${CRUBY_SRC}/prism
  ${CRUBY_SRC}/enc/unicode/17.0.0)

# NOTE:
# Scoped to C. Target-wide options reach the assembly source too, where
# -include malloc.h makes the assembler read a C header and report every
# declaration in it as an invalid instruction.
set(CRUBY_C_FLAGS
  -O1
  -fno-strict-overflow
  -fvisibility=hidden
  # CRuby is built against its own configure output upstream; against a
  # hand-written config.h it produces a lot of noise that says nothing about
  # this port.
  -Wall -Wno-unused-parameter -Wno-unused-value -Wno-parentheses
  -Wno-sign-compare -Wno-unused-function -Wno-tautological-compare
  -Wno-misleading-indentation -Wno-deprecated-declarations -Wno-undef
  # Several files call malloc without including the header, which the upstream
  # build gets away with because configure arranges it.
  -include malloc.h)

target_compile_options(ruby-static PRIVATE "$<$<COMPILE_LANGUAGE:C>:${CRUBY_C_FLAGS}>")

target_compile_definitions(ruby-static PRIVATE
  $<$<COMPILE_LANGUAGE:C>:RUBY_EXPORT>
  $<$<COMPILE_LANGUAGE:C>:DISABLE_RUBYGEMS=1>
  $<$<COMPILE_LANGUAGE:C>:DLEXT=".so">)

# The coroutine switcher is hand-written assembly. It needs no libc and no
# symbol prefix on this target.
#
# Passed as a compile option rather than through COMPILE_DEFINITIONS: CMake
# silently drops function-style macros from that property, with only a warning
# at generate time, and the assembly then fails to build.
set_source_files_properties(${CRUBY_SRC}/coroutine/amd64/Context.S PROPERTIES
  COMPILE_OPTIONS "-DPREFIXED_SYMBOL(name)=name")
